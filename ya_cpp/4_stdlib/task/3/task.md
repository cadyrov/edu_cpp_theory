#include <algorithm>
#include <iostream>

struct Point {
    int x = 0;
    int y = 0;
};

void GUI_Plot(Point* points, size_t n) {
    for (size_t i = 0; i < n; ++i) {
        printf("%d : %d\n", points[i].x, points[i].y);
    }
}
